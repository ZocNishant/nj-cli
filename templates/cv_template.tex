%% nj-cli CV template
%%
%% Contract: nj/tailoring/renderer.py::_fill_template does a literal string
%% replace of each double-percent-delimited token below. Every key in that dict
%% must appear here or its content is silently dropped from the PDF.
%%
%% NOTE: the substitution is a blind string replace over this entire file,
%% comments included. Never write a placeholder token in a comment — it will be
%% expanded into live LaTeX in the middle of the commented-out region.
%%
%% Two blocks emit their own \section header and must NOT be wrapped here: the
%% SUMMARY block (emits \section{Summary} plus its own itemize) and the GAP
%% block (emits its own \section plus its own subheading-list wrapper).
%% The rest emit bare content and are wrapped below.
%%
%% The list macros below are deliberately \begingroup/\endgroup rather than
%% itemize environments. The _render_* helpers emit nothing at all for an empty
%% section, and an empty itemize is a LaTeX error ("Something's wrong--perhaps a
%% missing \item"). A group is a no-op when empty, so a CV with no projects or
%% no certifications still compiles. \resumeItem draws its own bullet for the
%% same reason: it is emitted both inside a list (experience bullets) and bare
%% (gap, certifications, memberships).

%% ---------- macros the renderer's output depends on ----------

\newcommand{\resumeSubHeadingListStart}{\begingroup}
\newcommand{\resumeSubHeadingListEnd}{\endgroup\vspace{2pt}}
\newcommand{\resumeItemListStart}{\begingroup}
\newcommand{\resumeItemListEnd}{\endgroup\vspace{2pt}}

\newcommand{\resumeItem}[1]{%
  \par\noindent\hangindent=1.3em\hspace*{0.7em}\textbullet\hspace{0.5em}%
  \small #1\par\vspace{1pt}%
}

% \resumeSubheading{title}{dates}{company}{location}   (experience)
% \resumeSubheading{institution}{location}{degree}{dates}  (education)
\newcommand{\resumeSubheading}[4]{%
  \vspace{5pt}\par\noindent
  \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{#1} & \small #2 \\
    \textit{\small #3} & \textit{\small #4} \\
  \end{tabular*}\par\vspace{-2pt}%
}

\newcommand{\resumeProjectHeading}[2]{%
  \vspace{4pt}\par\noindent
  \begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
    #1 & \small #2 \\
  \end{tabular*}\par\vspace{-2pt}%
}
